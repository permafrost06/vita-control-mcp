cmake_minimum_required(VERSION 3.16)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  if(NOT DEFINED ENV{VITASDK})
    message(FATAL_ERROR "Set VITASDK to your VitaSDK installation")
  endif()
  set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "Vita toolchain")
endif()
project(vita_control C)
include("${VITASDK}/share/vita.cmake" REQUIRED)
add_executable(vita_control plugin/main.c plugin/core.c)
target_compile_options(vita_control PRIVATE -O2 -Wall -Wextra -Werror -std=gnu11)
target_link_options(vita_control PRIVATE -nostdlib -Wl,-e,module_start)
target_link_libraries(vita_control taihen_stub SceLibKernel_stub SceLibc_stub
  SceKernelThreadMgr_stub SceIofilemgr_stub SceSysmodule_stub SceNet_stub_weak gcc)
vita_create_self(vita_control.suprx vita_control CONFIG ${CMAKE_SOURCE_DIR}/plugin/exports.yml UNSAFE)
